let
 Source = Csv.Document(File.Contents("C:\PATH\TO\PROJECT\data\fact_flights.csv"),[Delimiter=",",Encoding=65001,QuoteStyle=QuoteStyle.Csv]),
 Headers = Table.PromoteHeaders(Source,[PromoteAllScalars=true]),
 Types = Table.TransformColumnTypes(Headers,{
  {"flight_id",type text},{"flight_date",type date},{"airline_id",type text},
  {"origin_airport",type text},{"destination_airport",type text},
  {"scheduled_departure",type datetime},{"actual_departure",type datetime},
  {"scheduled_arrival",type datetime},{"actual_arrival",type datetime},
  {"distance_km",Int64.Type},{"aircraft_capacity",Int64.Type},{"passengers",Int64.Type},
  {"delay_minutes",Int64.Type},{"cancelled_flag",Int64.Type},
  {"average_ticket_fare_inr",Currency.Type},{"flight_revenue_inr",Currency.Type},
  {"delay_reason",type text}}),
 CleanKeys = Table.TransformColumns(Types,{
  {"flight_id",Text.Trim,type text},{"origin_airport",Text.Trim,type text},
  {"destination_airport",Text.Trim,type text}}),
 AddRoute = Table.AddColumn(CleanKeys,"route",each [origin_airport]&" - "&[destination_airport],type text)
in AddRoute
